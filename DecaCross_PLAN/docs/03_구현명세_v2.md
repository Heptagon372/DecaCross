```kotlin
package kr.decacross.compat.model

// ── 버전 ────────────────────────────────────────────────
/**
 * MC 버전 비교의 유일한 기준.
 * # 불변식
 * - label 이나 SemVer 로 MC 버전을 비교하는 코드는 전부 버그다.
 *   2026년 Mojang 이 "1." 접두사를 폐지해 "1.21.8" 과 "26.3" 의 문자열 비교가 뒤집힌다.
 * - 한 번 발급된 값은 영원히 재할당되지 않는다.
 */
@JvmInline
value class McOrdinal(val value: Int) : Comparable<McOrdinal> {
    override fun compareTo(other: McOrdinal) = value.compareTo(other.value)
}

data class McVersion(
    val ordinal: McOrdinal,
    val label: String,                 // "1.8.9" | "1.21.8" | "26.3"
    val releasedAt: Instant,
    val isSnapshot: Boolean = false,
    val javaMin: Int,                  // 8 | 16 | 17 | 21 | 25
    val javaRecommended: Int,
    val rpFormat: PackFormat?,         // 리소스팩
    val dpFormat: PackFormat?,         // 데이터팩 ← 별개! 합치지 마라
    val nms: String? = null,           // "v1_21_R1"
    val protocol: Int? = null,
)

/** 1.21.9 부터 "88.0" 형태. Int 로 다루면 비교가 깨진다. */
data class PackFormat(val major: Int, val minor: Int = 0) : Comparable<PackFormat> {
    override fun compareTo(other: PackFormat) =
        compareValuesBy(this, other, PackFormat::major, PackFormat::minor)
    companion object { fun parse(s: String): PackFormat? = TODO() }
}

sealed interface PackDecl {
    data class Single(val format: PackFormat) : PackDecl
    data class Range(val min: PackFormat, val max: PackFormat) : PackDecl
    data class Supported(val formats: List<PackFormat>) : PackDecl

    fun isCompatibleWith(target: PackFormat): Boolean
}

// ── 코어 / 런타임 ───────────────────────────────────────
enum class CoreKey { PAPER, PURPUR, FOLIA, SPIGOT, VANILLA, FABRIC, NEOFORGE, FORGE }
enum class LoaderFamily { BUKKIT, FABRIC, FORGE, VANILLA }
enum class Channel { STABLE, EXPERIMENTAL }

data class CoreBuild(
    val core: CoreKey,
    val mc: McOrdinal,
    val build: String,
    val channel: Channel,
    val downloadUrl: String,
    val sha256: String,
    val size: Long,
)

enum class Distribution { TEMURIN }
enum class ImageType { JRE, JDK }
enum class Os { WINDOWS, MAC, LINUX }
enum class Arch { X64, AARCH64 }

data class JavaSpec(
    val feature: Int,
    val distribution: Distribution = Distribution.TEMURIN,
    val image: ImageType = ImageType.JRE,
)

// ── 콘텐츠 ──────────────────────────────────────────────
enum class ContentKind { PLUGIN, MOD, SCRIPT, RESOURCE_PACK, DATA_PACK, WORLD, MODEL_PACK }
enum class Source { MODRINTH, HANGAR, SPIGOT, URL, USER_UPLOAD, BUNDLED }

data class Content(
    val slug: String,
    val name: String,
    val kind: ContentKind,
    val source: Source,
    val license: String? = null,
    /**
     * # 불변식
     * false 면 우리가 파일을 미러링·재배포하지 않는다.
     * 라이선스를 확인하지 못했으면 반드시 false. 기본값이 안전한 쪽이어야 한다.
     */
    val redistributable: Boolean = false,
)

data class ContentVersion(
    val slug: String,
    val version: String,
    val fileUrl: String? = null,
    val sha256: String? = null,
    val size: Long? = null,
    /** ASM 으로 읽은 class major version 에서 역산한 실측값. 메타데이터보다 우선 신뢰. */
    val javaMajor: Int? = null,
    val apiVersion: String? = null,
    val loaders: Set<LoaderFamily> = emptySet(),
    val mcMin: McOrdinal? = null,
    val mcMax: McOrdinal? = null,
    val pack: PackDecl? = null,
    val deps: List<Dep> = emptyList(),
)

enum class DepKind { REQUIRE, OPTIONAL, PROVIDES }   // ★ CONFLICT 없음 — PROVIDES 로 표현

sealed interface DepTarget {
    data class Slug(val value: String) : DepTarget
    data class Cap(val value: Capability) : DepTarget
}

data class Dep(val kind: DepKind, val target: DepTarget, val range: String = "*")

/** 배타 제약을 표현하는 가상 패키지 키. 같은 Capability 제공자는 하나만 선택된다. */
sealed interface Capability {
    data object CustomItemFramework : Capability   // ItemsAdder | Oraxen | Nexo
    data object EconomyProvider : Capability
    data object PermissionProvider : Capability
    data object AntiCheat : Capability
    data object ChunkGenerator : Capability
    data class Other(val key: String) : Capability
}
```

### 2.1 McOrdinal 발급 규칙

```
초기 시딩 (1회만)
  version_manifest_v2.json 의 release 를 releasedAt 오름차순 정렬
  i번째 → ordinal = 1000 + i*10          (사이 9칸은 스냅샷/핫픽스 여유)

신규 버전
  ordinal = max(ordinal) + 10            ← append-only. 재할당 코드를 만들지 마라

스냅샷
  직전 release ordinal + 1..9
```

## 3. compat-engine 공개 API

```kotlin
package kr.decacross.compat

fun resolve(req: ResolveRequest, db: CompatDb): ResolveOutcome

data class ResolveRequest(
    val mc: McSelector = McSelector.Any,
    val core: CoreKey? = null,
    val wants: List<Want> = emptyList(),
    val os: Os,
    val arch: Arch,
    val ramMb: Int,
    val allowSnapshots: Boolean = false,
    val allowExperimental: Boolean = false,
    /** 런처/데몬이 이미 쓰고 있는 메모리(MB). 권장 RAM 계산에서 차감한다. (Kotlin 전환 신규) */
    val hostOverheadMb: Int = 0,
)

sealed interface McSelector {
    data class Exact(val label: String) : McSelector    // "1.21.8"
    data class Family(val prefix: String) : McSelector  // "1.21" → 1.21..1.21.8
    data object Latest : McSelector
    data object Any : McSelector                        // 엔진이 고름 — 입문자 경로
}

data class Want(
    val slug: String,
    val kind: ContentKind,
    val version: String? = null,
    /** true 면 이 항목을 제거하는 해결책은 제시하지 않는다. */
    val pinned: Boolean = false,
)

sealed interface ResolveOutcome {
    data class Ok(val plan: Plan) : ResolveOutcome
    data class Conflict(val explanation: Explanation) : ResolveOutcome
}

data class Plan(
    val mc: McVersion,
    val core: CoreBuild,
    val java: JavaSpec,
    val items: List<PlanItem>,
    val warnings: List<Warning> = emptyList(),
    val confidence: Confidence,
    val score: Float,
    val estimatedDownloadBytes: Long,
    val recommendedRamMb: Int,
)

data class PlanItem(
    val content: ContentVersion,
    /** 사용자가 고른 게 아니라 의존성으로 자동 추가됨. UI 에서 구분 표시할 것. */
    val autoAdded: Boolean = false,
    val reason: String? = null,          // "EssentialsX 가 필요로 함"
    val confidence: Confidence,
)

enum class Confidence { GREEN, YELLOW, ORANGE, RED }

// ── 실패 설명 ───────────────────────────────────────────
data class Explanation(
    val headlineKo: String,              // "26.3에서는 ProtocolLib을 쓸 수 없습니다."
    val causeChain: List<CauseNode>,     // 최소 충돌 집합만. 전체 트리 노출 금지
    /** # 불변식: 항상 1개 이상. 0개면 엔진 버그다. */
    val fixes: List<Fix>,
)

data class CauseNode(val textKo: String, val subject: Pkg? = null)

data class Fix(
    val labelKo: String,
    val action: FixAction,
    val sideEffectsKo: List<String> = emptyList(),
    val recommended: Boolean = false,
)

sealed interface FixAction {
    data class BumpContent(val slug: String, val to: String) : FixAction
    data class ChangeMc(val to: String) : FixAction
    data class ChangeCore(val to: CoreKey) : FixAction
    data class ChangeJava(val to: Int) : FixAction
    data class RemoveContent(val slugs: List<String>) : FixAction
    data class ReplaceContent(val from: String, val to: String) : FixAction
}

interface CompatDb {
    fun mcByLabel(label: String): McVersion?
    fun mcLatest(allowSnapshot: Boolean = false): McVersion?
    fun mcInFamily(prefix: String): List<McVersion>
    fun coreBuilds(core: CoreKey, mc: McOrdinal, stableOnly: Boolean = true): List<CoreBuild>
    fun content(slug: String): Content?
    fun contentVersions(slug: String): List<ContentVersion>
    fun providersOf(cap: Capability): List<String>
    fun confidenceOf(slug: String, version: String, mc: McOrdinal, core: CoreKey): Confidence
}
```

### 3.1 불변식 (테스트로 강제)

| # | 불변식 | 테스트 이름 |
|---|---|---|
| I1 | `Conflict.explanation.fixes` 는 항상 1개 이상 | `invariant_conflictAlwaysHasFix` |
| I2 | `Plan` 의 모든 `REQUIRE` 의존성이 `items` 에 존재 | `invariant_depsClosed` |
| I3 | 같은 `Capability` 제공자가 2개 이상 선택되지 않음 | `invariant_capabilityExclusive` |
| I4 | `plan.java.feature >= plan.mc.javaMin` | `invariant_javaFloor` |
| I5 | `recommendedRamMb` 계산에 `hostOverheadMb` 가 반영됨 | `invariant_ramAccountsHost` |
| I6 | 콘텐츠 30개 입력 시 `resolve()` < 50ms (JIT 예열 후) | `bench_resolve30` |

> Rust 안에 있던 "native ↔ wasm parity" 불변식은 **사라졌다.** 엔진이 하나뿐이라 성립할 수 없다. 대신 I5가 들어왔다.

## 4. `pubgrub-kt` — 포팅 명세 ★ 최대 비용

### 4.1 공개 API

```kotlin
package kr.decacross.pubgrub

interface VersionSet<V : Comparable<V>> {
    val isEmpty: Boolean
    fun contains(v: V): Boolean
    fun intersect(other: VersionSet<V>): VersionSet<V>
    fun union(other: VersionSet<V>): VersionSet<V>
    fun complement(): VersionSet<V>
    companion object {
        fun <V : Comparable<V>> empty(): VersionSet<V> = TODO()
        fun <V : Comparable<V>> full(): VersionSet<V> = TODO()
        fun <V : Comparable<V>> singleton(v: V): VersionSet<V> = TODO()
    }
}

interface DependencyProvider<P : Any, V : Comparable<V>> {
    /** 후보 중 하나 선택. 선호 순서대로 반환(STABLE 먼저, 그다음 최신). null 이면 해당 없음. */
    fun chooseVersion(pkg: P, range: VersionSet<V>): V?
    fun getDependencies(pkg: P, version: V): Dependencies<P, V>
    /** 높을수록 먼저 결정. 후보 수가 적은 패키지를 먼저 고르면 탐색이 줄어든다. */
    fun prioritize(pkg: P, range: VersionSet<V>): Int = 0
}

sealed interface Dependencies<P, V : Comparable<V>> {
    data class Available<P, V : Comparable<V>>(val map: Map<P, VersionSet<V>>) : Dependencies<P, V>
    data class Unavailable<P, V : Comparable<V>>(val reasonKo: String) : Dependencies<P, V>
}

fun <P : Any, V : Comparable<V>> resolve(
    provider: DependencyProvider<P, V>,
    root: P,
    rootVersion: V,
): SolverResult<P, V>

sealed interface SolverResult<P, V : Comparable<V>> {
    data class Solution<P, V : Comparable<V>>(val selected: Map<P, V>) : SolverResult<P, V>
    data class NoSolution<P, V : Comparable<V>>(val tree: DerivationTree<P, V>) : SolverResult<P, V>
}

sealed interface DerivationTree<P, V : Comparable<V>> {
    /** 외부 사실 — 사용자에게 보여줄 수 있는 것 */
    data class External<P, V : Comparable<V>>(val kind: ExternalKind<P, V>) : DerivationTree<P, V>
    /** 내부 유도 — 두 원인의 결합 */
    data class Derived<P, V : Comparable<V>>(
        val cause1: DerivationTree<P, V>,
        val cause2: DerivationTree<P, V>,
        val terms: Map<P, VersionSet<V>>,
    ) : DerivationTree<P, V>
}

sealed interface ExternalKind<P, V : Comparable<V>> {
    data class NotRoot<P, V : Comparable<V>>(val pkg: P, val v: V) : ExternalKind<P, V>
    data class NoVersions<P, V : Comparable<V>>(val pkg: P, val range: VersionSet<V>) : ExternalKind<P, V>
    data class Unavailable<P, V : Comparable<V>>(val pkg: P, val range: VersionSet<V>, val reasonKo: String) : ExternalKind<P, V>
    data class FromDependencyOf<P, V : Comparable<V>>(
        val pkg: P, val range: VersionSet<V>,
        val depPkg: P, val depRange: VersionSet<V>,
    ) : ExternalKind<P, V>
}
```

### 4.2 내부 구조 (PubGrub 알고리즘 그대로)

```
Incompatibility      "이 조합은 불가능하다"는 항들의 집합
PartialSolution      지금까지의 결정(decision) + 유도(derivation) 스택
unitPropagation()    현재 부분해에서 강제되는 결론 전파
conflictResolution() 충돌 시 원인 역추적 → 새 incompatibility 학습 + 백점프
decisionMaking()     다음 패키지·버전 선택 (prioritize 사용)
```

### 4.3 포팅 검증 전략

- **골든 케이스를 이식한다.** `pubgrub-rs` 의 테스트 스위트와 Dart `pub` 의 해결 테스트를 JSON 픽스처로 변환해 그대로 돌린다
- 속성 테스트: 무작위 의존성 그래프 생성 → 해가 있으면 실제로 모든 제약을 만족하는지 검증
- `OfflineDependencyProvider` 유사 구현을 테스트용으로 제공
- **DerivationTree 품질이 핵심이다.** 해만 찾고 트리가 엉망이면 포팅 실패다 — explain 이 못 돌아간다

> 기간 2~3주. **1주차에 착수해 6주차까지 끝낸다.** 7주차에 compat-engine 이 이걸 쓴다.

| 패키지 | 반환하는 의존성 | 대응 규칙 |
|---|---|---|
| `Root` | `Mc ∈ selector`, `Core ∈ *`, `Content(slug) ∈ 요청범위` | 사용자 요구 |
| `Core(paper)@build` | `Mc = 이 빌드의 MC`, `Java >= mc.javaMin`, `Api(BUKKIT) = 이 코어의 API 수준` | R1, R3 |
| `Content@ver` | `Api(kind) ∈ apiVersion 구간` | R4 |
| | `Java >= javaMajor` (ASM 실측), 알려진 상한 있으면 `Java < 상한` | R1', R2 |
| | `Mc ∈ [mcMin, mcMax]` | R3, R12 |
| | `Content(dep) ∈ range` (REQUIRE) | R6, R7 |
| | `Provides(cap) = Unit` (PROVIDES) | **R9 배타** |
| `Provides(cap)@Unit` | 없음 | — |

**배타가 왜 이걸로 되는가:** `Provides(CustomItemFramework)` 는 버전이 `Unit` 하나다. PubGrub 은 한 패키지에서 버전 하나만 고르므로, ItemsAdder 와 Oraxen 이 모두 이 가상 패키지를 요구하면 서로 다른 제공자를 통한 요구가 충돌로 잡힌다. Debian 의 `Provides`/`Conflicts` 관용구와 같다.

### 5.1 explain — DerivationTree → 한국어

```kotlin
fun explain(tree: DerivationTree<Pkg, Ver>, ctx: ExplainCtx): Explanation
```

```
① 후위 순회하며 External 노드만 수집 (Derived 는 경로 추적용)
② 같은 subject 노드 병합 → 최소 충돌 집합
③ ko 템플릿으로 문장 생성
④ 충돌 축(mc / java / core / content)마다 Fix 후보 생성
⑤ ★ 각 Fix 를 가정하고 resolve() 를 실제로 재실행 → 진짜 되는 것만 남긴다
⑥ 부수효과(함께 빠지는 항목) 계산 → sideEffectsKo
```

## 6. `jvm-analysis` — Kotlin 전환의 핵심 이득 ★

### 6.1 StaticVerify (K-1) — VERIFIER 를 80배 빠르게

```kotlin
package kr.decacross.analysis

data class StaticVerifyResult(
    val ok: Boolean,
    val missingClasses: List<MissingRef>,
    val missingMethods: List<MissingRef>,
    val warnings: List<String>,
)

data class MissingRef(val from: String, val target: String, val detail: String)

/**
 * 플러그인 jar 의 모든 외부 참조를 코어 API jar 기준으로 해석한다.
 * 서버를 띄우지 않고 NoClassDefFoundError / NoSuchMethodError 를 예측한다.
 *
 * 조합 1건당 목표: 500ms 이내 (실제 기동은 40초)
 */
fun staticVerify(
    pluginJar: Path,
    coreApiJars: List<Path>,      // paper-api, bukkit, 그리고 함께 설치될 다른 플러그인들
    javaFeature: Int,
): StaticVerifyResult
```

구현 요지:
1. ASM `ClassReader` 로 각 `.class` 순회
2. `ClassVisitor` / `MethodVisitor` 로 참조하는 외부 타입·메서드 시그니처 수집
3. classpath(코어 API + 동시 설치 플러그인 + JDK)에서 해석 시도
4. 미해결 참조를 보고. **단, 리플렉션·`softdepend` 로 감싼 참조는 경고로 낮춘다** (오탐 방지)

> **오탐 관리가 관건이다.** 플러그인은 `try { Class.forName(...) }` 로 선택적 의존을 자주 쓴다.
> 미해결 참조가 `try/catch` 블록 안이거나 `softdepend` 에 선언된 대상이면 **경고**, 아니면 **에러**로 분류한다.

**보안 (테스트로 강제):**
- [ ] `Origin` 화이트리스트(`https://decacross.kr`, 개발 시 `http://localhost:3000`) 아니면 403
- [ ] `Host` 헤더가 `127.0.0.1:{port}` 정확히 아니면 403 (DNS rebinding)
- [ ] `X-Decacross-Bridge: 1` 없으면 403 (단순요청 차단 → preflight 강제)
- [ ] `/ping` 응답에 서버 목록·경로·사용자명이 **절대** 포함되지 않음
- [ ] `/install` 은 사용자가 네이티브 다이얼로그에서 승인하기 전에 파일을 쓰지 않음
- [ ] 브리지 라우트에서 UI 라우트 핸들러를 재사용하지 않음 (실수로 정보가 새는 경로)

## 8. 런처 JVM vs 서버 JVM — 반드시 분리 ★ Kotlin 전환 신규 함정

```
%LOCALAPPDATA%\Decacross\
├─ jre\                       ← jlink 로 만든 런처 전용 커스텀 런타임 (번들)
└─ runtimes\
   ├─ temurin-8-jre\          ← 서버 실행용 (Adoptium 다운로드)
   ├─ temurin-17-jre\
   ├─ temurin-21-jre\
   └─ temurin-25-jre\
```

**절대 번들 JRE 로 마크 서버를 실행하지 마라.** 두 가지 이유:
1. jlink 커스텀 런타임은 모듈이 빠져 있어 서버가 안 돌 수 있다
2. 버전이 우연히 맞아도, 다음 런처 업데이트에서 바뀌면 서버가 갑자기 깨진다

feature 가 같아 보여도 **항상 별도로 받는다.**

### 8.1 메모리 예산

| 프로세스 | 힙 상한 | 비고 |
|---|---|---|
| `decacross-daemon` | `-Xmx192m` | 상주. UI 닫아도 유지 |
| `decacross-ui` | `-Xmx384m` | 창 닫으면 종료 |
| 마크 서버 | 사용자 지정 | 별도 JVM — 위 둘과 무관 |

`recommendedRamMb` 계산:
```
권장 = min(시스템메모리 × 0.5, 상한) − hostOverheadMb − 클라이언트예약(기본 2048)
hostOverheadMb = 데몬 실사용 + (UI 실행 중이면 UI 실사용)
```

## 13. 절대 하지 말 것

1. MC 버전을 **문자열·SemVer 로 비교** → `McOrdinal` 만 쓴다
2. `ordinal` **재할당** → append-only
3. `pack_format` 을 **Int 로 저장** → `numeric` + `PackFormat(major, minor)`
4. 리소스팩·데이터팩 포맷 **한 컬럼에 합치기**
5. `redistributable` 없이 콘텐츠 저장
6. **번들 JRE 로 마크 서버 실행** ← Kotlin 전환 신규 함정
7. 시스템 `JAVA_HOME` / `PATH` / 레지스트리 수정
8. 생성된 `start.bat` 이 런처 의존하게 만들기
9. 사용자 폴더에 **반쯤 만든 서버** 남기기 → 스테이징 + 원자적 이동
10. `stop` 없이 프로세스 **kill**
11. 로그를 **라인마다** WS 전송 → 50ms 배칭
12. **브리지 라우트에서 UI 핸들러 재사용** ← 정보 유출 경로
13. `Explanation` 을 **fix 없이** 반환
14. 외부 API 를 **generic User-Agent** 로 호출 (PaperMC 정책 위반)
15. `core/*` 모듈에 **파일·네트워크 I/O** 넣기
16. 정적 검증 미해결 참조를 **전부 에러로** 처리 → 리플렉션·softdepend 는 경고
